### Supabase Table Created
```sql
CREATE TABLE client_portals (
  id UUID PRIMARY KEY,
  client_name TEXT,
  client_email TEXT,
  development TEXT,
  lot_number INTEGER,
  house_model TEXT,
  access_code TEXT,
  hero_image_url TEXT,
  status TEXT DEFAULT 'active',
  portal_url TEXT,
  documents JSONB DEFAULT '[]',
  timeline JSONB DEFAULT '[]',
  notes TEXT,
  created_at TIMESTAMPTZ,
  updated_at TIMESTAMPTZ
);
```
